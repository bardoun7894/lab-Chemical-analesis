التعريفات الأساسية
Ladle ID
معرّف المعلقة (الباتش) — مجموعة المواسير المنتجة من صهرة واحدة في مرحلة Melting.
Pipe Order
رقم ترتيب الماسورة داخل الـ Ladle. يُستخدم لتحديد First و Last.
First Pipe ⭐
الماسورة ذات min(Pipe Order) داخل نفس الـ Ladle ID.
Last Pipe ⭐
الماسورة ذات max(Pipe Order) داخل نفس الـ Ladle ID.
Decision Type
نوع قرار الاختبار المعيّن للمعلقة من التحليل الكيميائي — يحدد أي مواسير تحتاج اختباراً ميكانيكياً.
Sample Pipe ⭐
الماسورة المحددة للاختبار الميكانيكي الفعلي. تُعلَّم بنجمة في شاشة المعمل.
Cascade Decision
تعميم قرار العينة على كل مواسير نفس الـ Ladle تلقائياً بعد ظهور النتيجة.
WAITING
حالة المواسير غير العينات — تنتظر نتيجة العينة. لا قرار نهائي حتى تكتمل الاختبارات.
القاعدة الأساسية لتحديد First / Last:
First = الماسورة التي لها min(pipe_order) في نفس ladle_id
Last = الماسورة التي لها max(pipe_order) في نفس ladle_id
2
أنواع القرارات (Decision Types)
🚫 Rejected
الباتش مرفوض كيميائياً أو ميكانيكياً.
كل المواسير → BLOCK
لا اختبارات ميكانيكية. لا مراحل إنتاجية.
⭐ Last Pipe Only
آخر ماسورة فقط (max order) تُختبر ميكانيكياً.
نتيجتها تُعمَّم Cascade على كل مواسير الـ Ladle.
الباقي في WAITING حتى تظهر النتيجة.
⭐⭐ First & Last
أول (min) + آخر (max) ماسورة تُختبران.
لو الاثنان Pass → الجميع ACCEPT (Cascade)
أي منهما Fail → الباقي HOLD (Cascade)
✅ 100% Pipes
كل ماسورة تُختبر بشكل مستقل. لا تعميم.
Pass → ACCEPT هذه الماسورة فقط
Fail → REJECT هذه الماسورة فقط

# ═══════════════════════════════════════════════════════════════════════
# GCP QC — Pipe Decision Engine
# Mechanical Lab Stage | Ductile Iron
# ═══════════════════════════════════════════════════════════════════════

from typing import List, Dict, Literal
from dataclasses import dataclass, field

DecisionType = Literal["Rejected", "Last Pipe Only", "First & Last", "100% Pipes"]
MechResult   = Literal["Pass", "Fail", "WAIT"]
PipeDecision = Literal["ACCEPT", "REJECT", "HOLD", "WAITING", "BLOCK", ""]


# ── Data Model ──────────────────────────────────────────────────────────
@dataclass
class Pipe:
    pipe_code:    str
    ladle_id:     str
    pipe_order:   int
    mech_result:  MechResult = "WAIT"
    # Computed fields
    role:         str = field(default="", init=False)
    is_sample:    bool = field(default=False, init=False)
    decision:     PipeDecision = field(default="", init=False)
    cascade_from: str = field(default="", init=False)


# ═══════════════════════════════════════════════════════════════════════
# STEP 1 — تحديد الأدوار (First / Last) بناءً على Pipe Order
# ═══════════════════════════════════════════════════════════════════════
def assign_roles(pipes: List[Pipe]) -> None:
    """
    First = min(pipe_order) per ladle_id
    Last  = max(pipe_order) per ladle_id
    يُنفَّذ تلقائياً قبل أي عملية قرار.
    """
    ladles: Dict[str, List[Pipe]] = {}
    for p in pipes:
        ladles.setdefault(p.ladle_id, []).append(p)

    for group in ladles.values():
        min_ord = min(p.pipe_order for p in group)
        max_ord = max(p.pipe_order for p in group)
        for p in group:
            is_first = p.pipe_order == min_ord
            is_last  = p.pipe_order == max_ord
            if   is_first and is_last: p.role = "First & Last"
            elif is_first:              p.role = "First"
            elif is_last:               p.role = "Last"
            else:                       p.role = "Any"


# ═══════════════════════════════════════════════════════════════════════
# STEP 2 — تحديد العينات (⭐ is_sample)
# ═══════════════════════════════════════════════════════════════════════
def mark_samples(pipes: List[Pipe], dt: DecisionType) -> None:
    """يضع is_sample=True على المواسير التي تحتاج اختباراً ميكانيكياً."""
    for p in pipes:
        match dt:
            case "Rejected":
                p.is_sample = False
            case "100% Pipes":
                p.is_sample = True
            case "Last Pipe Only":
                p.is_sample = p.role in ("Last", "First & Last")
            case "First & Last":
                p.is_sample = p.role in ("First", "Last", "First & Last")


# ═══════════════════════════════════════════════════════════════════════
# STEP 3 — محرك القرار مع Cascade per Ladle
# ═══════════════════════════════════════════════════════════════════════
def run_decision_engine(pipes: List[Pipe], dt: DecisionType) -> None:
    """يحسب p.decision لكل ماسورة ويطبّق cascade داخل كل Ladle."""
    ladles: Dict[str, List[Pipe]] = {}
    for p in pipes:
        ladles.setdefault(p.ladle_id, []).append(p)
    for group in ladles.values():
        _decide_ladle(group, dt)


def _decide_ladle(group: List[Pipe], dt: DecisionType) -> None:

    # ── REJECTED ──────────────────────────────────────────────────────
    if dt == "Rejected":
        for p in group:
            p.decision, p.cascade_from = "BLOCK", "Ladle Rejected"
        return

    # ── 100% PIPES ────────────────────────────────────────────────────
    if dt == "100% Pipes":
        for p in group:
            p.decision = ("ACCEPT"  if p.mech_result == "Pass"
                          else "REJECT"  if p.mech_result == "Fail"
                          else "WAITING")
            p.cascade_from = "individual"
        return

    # ── LAST PIPE ONLY ────────────────────────────────────────────────
    if dt == "Last Pipe Only":
        last   = next((p for p in group if p.role in ("Last", "First & Last")), None)
        batch  = ("ACCEPT"  if last and last.mech_result == "Pass"
                  else "HOLD"   if last and last.mech_result == "Fail"
                  else "WAITING")
        source = last.pipe_code if last else "?"
        for p in group:
            p.decision     = batch
            p.cascade_from = ("sample" if p.pipe_code == source
                              else f"cascade from {source}")
        return

    # ── FIRST & LAST ──────────────────────────────────────────────────
    if dt == "First & Last":
        first = next((p for p in group if p.role in ("First", "First & Last")), None)
        last  = next((p for p in group if p.role in ("Last",  "First & Last")), None)
        f_res = first.mech_result if first else "WAIT"
        l_res = last.mech_result  if last  else "WAIT"

        both_ready = f_res != "WAIT" and l_res != "WAIT"
        both_pass  = f_res == "Pass" and l_res == "Pass"

        for p in group:
            is_sample = p.role in ("First", "Last", "First & Last")

            if is_sample:
                # 1. قرار فردي أولاً
                p.decision = ("ACCEPT" if p.mech_result == "Pass"
                              else "HOLD"   if p.mech_result == "Fail"
                              else "WAITING")
                # 2. override: لو الاثنان Pass → الجميع ACCEPT
                if both_ready and both_pass:
                    p.decision = "ACCEPT"
                p.cascade_from = "sample"
            else:
                # باقي المواسير — Cascade
                p.decision = ("WAITING" if not both_ready
                              else "ACCEPT"  if both_pass
                              else "HOLD")
                p.cascade_from = f"First({f_res})+Last({l_res})"


# ═══════════════════════════════════════════════════════════════════════
# MAIN ENTRY — دالة واحدة تُستدعى من الـ API
# ═══════════════════════════════════════════════════════════════════════
def process_ladle_batch(
    pipes: List[Pipe],
    decision_type: DecisionType
) -> List[Dict]:
    """
    الدالة الرئيسية — تُستدعى بعد كل إدخال نتيجة ميكانيكية.
    تُعيد قائمة dicts جاهزة للحفظ في الـ DB.
    """
    assign_roles(pipes)
    mark_samples(pipes, decision_type)
    run_decision_engine(pipes, decision_type)

    return [
        {
            "pipe_code":    p.pipe_code,
            "ladle_id":     p.ladle_id,
            "pipe_order":   p.pipe_order,
            "role":         p.role,
            "is_sample":    p.is_sample,
            "mech_result":  p.mech_result,
            "decision":     p.decision,
            "cascade_from": p.cascade_from,
        }
        for p in pipes
    ]

if __name__ == "__main__":

    # مثال 1: First & Last — 5 مواسير — كلاهما Pass
    batch1 = [

    # طباعة النتائج
